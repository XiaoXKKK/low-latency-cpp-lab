cmake_minimum_required(VERSION 3.20)
project(low_latency_cpp_lab VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(LAB_SANITIZER "none" CACHE STRING "none, address, or thread (Debug only)")
option(LAB_NATIVE "Enable -march=native; reduces portability" OFF)
if(NOT LAB_SANITIZER STREQUAL "none")
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Sanitizers require a single-config Debug build")
  endif()
  if(LAB_SANITIZER STREQUAL "address")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  elseif(LAB_SANITIZER STREQUAL "thread")
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  else()
    message(FATAL_ERROR "Unknown LAB_SANITIZER")
  endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(LAB_NATIVE)
  add_compile_options(-march=native)
endif()
option(LAB_LTO "Enable interprocedural optimization" OFF)
if(LAB_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT lab_ipo_supported OUTPUT lab_ipo_error)
  if(NOT lab_ipo_supported)
    message(FATAL_ERROR "IPO unavailable: ${lab_ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
find_package(Threads REQUIRED)
add_library(lab_core src/benchmark.cpp src/results.cpp src/linux_memory.cpp src/network.cpp)
target_include_directories(lab_core PUBLIC include)
target_link_libraries(lab_core PUBLIC Threads::Threads)
target_compile_definitions(lab_core PUBLIC _GNU_SOURCE LAB_BUILD_TYPE="${CMAKE_BUILD_TYPE}" LAB_SANITIZER="${LAB_SANITIZER}")
add_executable(lab_bench src/main.cpp benchmarks/timer/timer_overhead.cpp
  benchmarks/cache/sequential_vs_random.cpp benchmarks/cache/false_sharing.cpp
  benchmarks/branch/branch_prediction.cpp benchmarks/memory/malloc_vs_pool.cpp
  benchmarks/synchronization/mutex_vs_spinlock.cpp benchmarks/synchronization/spsc_ring_buffer.cpp
  benchmarks/scheduling/thread_affinity.cpp
  benchmarks/timer/interval_timing.cpp benchmarks/cache/cache_patterns.cpp
  benchmarks/memory/allocator_batch.cpp benchmarks/memory/page_behavior.cpp
  benchmarks/synchronization/atomic_order.cpp benchmarks/synchronization/rw_locks.cpp
  benchmarks/scheduling/wait_and_arrival.cpp benchmarks/numa/numa_access.cpp benchmarks/network/loopback.cpp)
target_link_libraries(lab_bench PRIVATE lab_core)
# Keep the branch experiment's conditional jump observable. Inspect assembly too.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set_source_files_properties(benchmarks/branch/branch_prediction.cpp PROPERTIES COMPILE_OPTIONS "-fno-if-conversion;-fno-if-conversion2;-fno-tree-vectorize")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set_source_files_properties(benchmarks/branch/branch_prediction.cpp PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
endif()
enable_testing()
add_executable(lab_tests tests/test_core.cpp)
target_link_libraries(lab_tests PRIVATE lab_core)
add_test(NAME core COMMAND lab_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME cli_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:lab_bench>)
set_tests_properties(core cli_contract PROPERTIES TIMEOUT 90)
add_test(NAME perf_reporting COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_perf.py)
add_executable(lab_phase2_tests tests/test_phase2.cpp)
target_link_libraries(lab_phase2_tests PRIVATE lab_core)
add_test(NAME phase2_core COMMAND lab_phase2_tests)
add_test(NAME phase2_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_phase2_cli.py $<TARGET_FILE:lab_bench>)
set_tests_properties(phase2_core phase2_cli PROPERTIES TIMEOUT 120)
add_executable(lab_network_tests tests/test_network.cpp)
target_link_libraries(lab_network_tests PRIVATE lab_core)
add_test(NAME network_core COMMAND lab_network_tests)
add_test(NAME network_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_network_cli.py $<TARGET_FILE:lab_bench>)
set_tests_properties(network_core network_cli PROPERTIES TIMEOUT 90)
